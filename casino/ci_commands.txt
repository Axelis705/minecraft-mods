# (los crupieres de prueba flotan: si no, caen del y=120 y pueden morir antes de la comprobacion cuando el servidor del CI va lento)
summon casino:dealer 0 120 0 {NoGravity:1b}
setblock 2 119 2 casino:blackjack[facing=north,cell=2]
setblock 0 119 2 casino:blackjack[facing=north,cell=0]

execute if entity @e[type=casino:dealer] run say PRUEBA_CRUPIER_OK

# Pruebas en el servidor real: la máquina (dos mitades) y la vitrina se colocan y sus entidades de bloque existen
forceload add 0 0 15 15
setblock 0 99 0 minecraft:stone
setblock 3 99 0 minecraft:stone
setblock 0 100 0 autos_coleccion:maquina_expendedora[facing=south,half=lower]
setblock 0 101 0 autos_coleccion:maquina_expendedora[facing=south,half=upper]
execute if block 0 100 0 autos_coleccion:maquina_expendedora[half=lower] if block 0 101 0 autos_coleccion:maquina_expendedora[half=upper] run say PRUEBA_AUTOS_MAQUINA
setblock 3 100 0 autos_coleccion:vitrina[facing=south]
data merge block 3 100 0 {car:{id:"autos_coleccion:auto_corona_oro",Count:1b},spin:1b}
execute if data block 3 100 0 car run say PRUEBA_AUTOS_VITRINA
